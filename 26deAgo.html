var agora = new Date(); var horaBrasil = agora.toLocaleString("pt-BR", { timeZone: "America/Sao_Paulo", hour: "2-digit", hour12: false }); // Converte o texto da hora em um número inteiro var hora = parseInt(horaBrasil); console.log(`Agora são ${hora} horas`); if (hora < 12) { console.log('Bom dia'); } else if (hora <= 18) { console.log('Boa tarde'); } else { console.log('Boa Noite'); }
